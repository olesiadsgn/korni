function tg(method, params) {
  var r = UrlFetchApp.fetch('https://api.telegram.org/bot' + token() + '/' + method, {
    method: 'post', contentType: 'application/json', payload: JSON.stringify(params || {}), muteHttpExceptions: true});
  var j = JSON.parse(r.getContentText());
  if (!j.ok) throw new Error(method + ': ' + r.getContentText());
  return j.result;
}

}

    var st = getJ('state', {offset: 0, admin: null, pending: {}, mode: {}, last: {}, mute: []});
    var allow = getJ('allow', {});
    var upd = tg('getUpdates', {offset: st.offset, timeout: 0, allowed_updates: ['message', 'callback_query']});
    var news = [];
    upd.forEach(function (u) {
      st.offset = u.update_id + 1;
      try { handle(u, st, allow, news); } catch (e) { console.log('update fail', e); }
    });
    setJ('state', st); setJ('allow', allow);
    if (news.length) notify(news);
  } finally { lock.releaseLock(); }
}

function handle(u, st, allow, news) {
  if (u.callback_query) {
    var q = u.callback_query, uid0 = String(q.from.id);
    if (uid0 !== String(st.admin)) return;
    var parts = q.data.split(':'), target = parts[1], p = st.pending[target]; delete st.pending[target];
    var name = p ? p.name : target, txt;
    if (parts[0] === 'ok') { allow[target] = {name: name}; say(target, 'Олеся тебя добавила!'); welcome(target); txt = 'Пустила: ' + name; }
    else txt = 'Отклонила: ' + name;
    try { tg('answerCallbackQuery', {callback_query_id: q.id, text: txt}); tg('editMessageText', {chat_id: q.message.chat.id, message_id: q.message.message_id, text: txt}); } catch (e) {}
    return;
  }
  var m = u.message; if (!m || m.chat.type !== 'private') return;
  var frm = m.from, uid = String(frm.id), text = (m.text || '').trim();

  if ((frm.username || '').toLowerCase() === ADMIN_USER && String(st.admin) !== uid) {
    st.admin = uid; allow[uid] = {name: who(frm), admin: true};
    say(uid, 'Привет, Олеся! Ты админ. Когда кто-то новый нажмёт «Старт», я пришлю тебе кнопки «Пустить» и «Отклонить».');
    welcome(uid); if (text.indexOf('/start') === 0) return;
  }
  if (!allow[uid]) {
    if (!st.pending[uid]) {
      st.pending[uid] = {name: who(frm)};
      say(uid, 'Привет! Это семейный бот книги «Корни семьи». Олеся скоро тебя добавит, и тогда можно будет присылать фото.');
      if (st.admin) say(st.admin, 'Хочет войти: ' + who(frm) + '\nid ' + uid, {reply_markup: {inline_keyboard: [[{text: 'Пустить', callback_data: 'ok:' + uid}, {text: 'Отклонить', callback_data: 'no:' + uid}]]}});
    }
    return;
  }
  var payload = text.indexOf('/start ') === 0 ? text.slice(7).trim() : '';
  if (payload.indexOf('a_') === 0) {
    var rest = payload.slice(2), cut = rest.lastIndexOf('_'), pid = rest.slice(0, cut), idx = rest.slice(cut + 1);
    var qq = QUESTIONS.filter(function (x) { return x.id === pid && String(x.i) === idx; })[0];
    if (qq) { st.mode[uid] = {kind: 'answer', about: qq.who, question: qq.q, person: pid};
      say(uid, 'О ком: ' + qq.who + '\nВопрос: ' + qq.q + '\n\nРасскажите, что знаете. Можно текстом или голосовым, можно несколькими сообщениями.', {reply_markup: KB}); return; }
  }
  if (payload.indexOf('s_') === 0 && PEOPLE[payload.slice(2)]) {
    var sp = payload.slice(2); st.mode[uid] = {kind: 'story', about: PEOPLE[sp].full, person: sp};
    say(uid, 'История о человеке: ' + PEOPLE[sp].full + '.\n\nЕсли вы написали текст на сайте, он уже скопирован: просто вставьте его сюда и отправьте. Или расскажите голосом.', {reply_markup: KB}); return;
  }
  if (text.indexOf('/start') === 0 || text === '/help') { welcome(uid); return; }
  if (text === '/birthdays') {
    var i = st.mute.indexOf(uid);
    if (i >= 0) { st.mute.splice(i, 1); say(uid, 'Напоминания о днях рождения снова включены.', {reply_markup: KB}); }
    else { st.mute.push(uid); say(uid, 'Хорошо, больше не буду присылать напоминания о днях рождения. Включить снова: /birthdays', {reply_markup: KB}); }
    return;
  }
  if (text === '/fact' || text === B_FACT) { say(uid, fact(), {reply_markup: KB}); return; }
  if (text === B_BOOK) { say(uid, 'Вот наша книга. На первой странице нажмите на себя, и везде будет подписано, кто вам кто.', {reply_markup: {inline_keyboard: [[{text: 'Открыть книгу', url: SITE}]]}}); return; }
  if (text === B_PHOTO) { st.mode[uid] = {kind: 'photo'}; say(uid, 'Отправьте фото, как обычно в Телеграме. Можно сразу несколько.\n\nИ подпишите, если помните: кто на фото, где и примерно какой год. Можно рассказать голосом.', {reply_markup: KB}); return; }
  if (text === B_STORY) { st.mode[uid] = {kind: 'story'}; say(uid, 'Нажмите на микрофон и расскажите. О ком история, что случилось, примерно когда.\n\nМожно долго и несколькими голосовыми подряд. Можно и текстом.', {reply_markup: KB}); return; }
  if (text === B_FIX) { st.mode[uid] = {kind: 'fix'}; say(uid, 'Напишите или скажите голосом, что в книге не так. Например: «У бабушки Тони неправильная дата» или «На фото не Таня, а Света».\n\nМожно прислать скриншот страницы.', {reply_markup: KB}); return; }
  if (text === B_Q) {
    var r = QUESTIONS[Math.floor(Math.random() * QUESTIONS.length)];
    st.mode[uid] = {kind: 'answer', about: r.who, question: r.q, person: r.id};
    say(uid, 'Вот что мы пока не знаем.\n\nО ком: ' + r.who + '\nВопрос: ' + r.q + '\n\nЕсли знаете, ответьте текстом или голосовым. Если нет, нажмите «Ответить на вопрос» ещё раз, будет другой вопрос.', {reply_markup: KB});
    return;
  }

  // сохраняем присланное
  var d = new Date(m.date * 1000), md = st.mode[uid] || {};
  var base = Utilities.formatDate(d, 'Europe/Moscow', 'yyyy-MM-dd_HHmmss') + '_' + (allow[uid].name || uid).split(' ')[0] + (md.person ? '_' + md.person : '');
  var file = null, kind = 'текст', note = '';
  try {
    if (m.photo) { file = saveFile(m.photo[m.photo.length - 1].file_id, base + '.jpg', d); kind = 'фото'; }
    else if (m.document) { if ((m.document.file_size || 0) > 20e6) note = 'Файл больше 20 МБ, бот не может его скачать'; else { file = saveFile(m.document.file_id, base + '_' + (m.document.file_name || 'file'), d); kind = 'файл'; } }
    else if (m.voice || m.audio) { file = saveFile((m.voice || m.audio).file_id, base + '.ogg', d); kind = 'голосовое'; }
    else if (m.video || m.video_note) { var v = m.video || m.video_note; if ((v.file_size || 0) > 20e6) { note = 'Видео больше 20 МБ'; say(uid, 'Видео слишком большое для бота (больше 20 МБ). Пришлите, пожалуйста, покороче или напишите Олесе.'); } else { file = saveFile(v.file_id, base + '.mp4', d); kind = 'видео'; } }
    else if (!text) return;
  } catch (e) { note = 'Не скачалось: ' + e; }
  var typeName = {photo: 'фото', story: 'рассказ', fix: 'неточность', answer: 'ответ на вопрос'}[md.kind] || '';
  var caption = (typeName ? '[' + typeName + (md.question ? ': ' + md.question : '') + '] ' : '') + (m.caption || text || '') + (note ? ' (' + note + ')' : '');
  journal([Utilities.formatDate(d, 'Europe/Moscow', 'yyyy-MM-dd HH:mm'), allow[uid].name, file ? file.getUrl() : '', caption, '', md.about || '', '', '', 'новое', '']);
  news.push(allow[uid].name + ': ' + kind + (typeName ? ' (' + typeName + ')' : '') + (caption ? ' ' + caption.slice(0, 200) : ''));
  var now = Date.now();
  if (now - (st.last[uid] || 0) > 120000) say(uid, md.kind === 'fix' ? 'Спасибо, что заметили! Олеся поправит.' : 'Спасибо, всё получено! Олеся посмотрит и добавит в книгу.', {reply_markup: KB});
  st.last[uid] = now;
}

function notify(news) {
  var to = props().getProperty('NOTIFY_EMAIL') || Session.getEffectiveUser().getEmail();
  MailApp.sendEmail(to, 'Корни семьи: новое от родных (' + news.length + ')',
    'В семейный бот пришло новое:\n\n' + news.map(function (x) { return '• ' + x; }).join('\n') +
    '\n\nФайлы: https://drive.google.com/drive/folders/' + INBOX_FOLDER +
    '\nЖурнал: https://docs.google.com/spreadsheets/d/' + JOURNAL +
    '\n\nВечером Claude разберёт всё и пришлёт, что добавить в книгу.');
}

function years(n) { var a = n % 10, b = n % 100; return n + ' ' + (a === 1 && b !== 11 ? 'год' : (a >= 2 && a <= 4 && (b < 10 || b >= 20)) ? 'года' : 'лет'); }
function birthdays() {
  var now = new Date(), dm = Utilities.formatDate(now, 'Europe/Moscow', 'dd.MM'), y = +Utilities.formatDate(now, 'Europe/Moscow', 'yyyy');
  var day = +dm.slice(0, 2), mon = +dm.slice(3) - 1, msgs = [];
  BIRTHS.forEach(function (b) {
    if (b.dm !== dm) return;
    if (b.memory) msgs.push('📖 В этот день, ' + day + ' ' + MONTHS[mon] + ' ' + b.y + ' года, ' + (b.g === 'f' ? 'родилась ' : 'родился ') + b.full + '.' + (b.note ? ' ' + b.note : '') + '\n\nИстория в книге: ' + SITE + '#p-' + b.id);
    else msgs.push('🎂 Сегодня день рождения! ' + b.full + ', ' + b.short + '. Исполняется ' + years(y - b.y) + '.\n\nНе забудьте поздравить!');
  });
  if (!msgs.length) return;
  var st = getJ('state', {mute: []}), allow = getJ('allow', {});
  Object.keys(allow).forEach(function (uid) { if ((st.mute || []).indexOf(uid) >= 0) return; msgs.forEach(function (t) { say(uid, t); }); });
}
