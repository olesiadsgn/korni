// ===== Store.js =====
/* Память бота в свойствах скрипта.
   state: offset Телеграма, id админа, кто ждёт допуска, режим каждого человека, когда благодарили, кто выключил напоминания.
   allow: кому можно писать боту. */

function props() { return PropertiesService.getScriptProperties(); }
function getJ(k, d) { var v = props().getProperty(k); return v ? JSON.parse(v) : d; }
function setJ(k, v) { props().setProperty(k, JSON.stringify(v)); }

function loadState() {
  var st = getJ('state', {});
  ['pending', 'mode', 'last'].forEach(function (k) { st[k] = st[k] || {}; });
  st.mute = st.mute || []; st.offset = st.offset || 0;
  return st;
}
function loadAllow() { return getJ('allow', {}); }

// ===== Telegram.js =====
/* Запросы к Telegram Bot API. */

function token() {
  var t = props().getProperty('TG_TOKEN');
  if (!t) throw new Error('Нет TG_TOKEN в свойствах скрипта');
  return t;
}

function tg(method, params) {
  var r = UrlFetchApp.fetch('https://api.telegram.org/bot' + token() + '/' + method, {
    method: 'post', contentType: 'application/json', payload: JSON.stringify(params || {}), muteHttpExceptions: true});
  var j = JSON.parse(r.getContentText());
  if (!j.ok) throw new Error(method + ': ' + r.getContentText());
  return j.result;
}

/* Отправить текст. Ошибку только записываем: один упавший ответ не должен ронять весь разбор. */
function say(chat, text, extra) {
  var p = {chat_id: chat, text: text};
  for (var k in (extra || {})) p[k] = extra[k];
  try { return tg('sendMessage', p); } catch (e) { console.log(e); }
}
function sayKb(chat, text) { return say(chat, text, {reply_markup: KB}); }

function who(u) {
  return [u.first_name, u.last_name].filter(Boolean).join(' ') + (u.username ? ' (@' + u.username + ')' : '');
}

function downloadFile(fileId, name) {
  var f = tg('getFile', {file_id: fileId});
  return UrlFetchApp.fetch('https://api.telegram.org/file/bot' + token() + '/' + f.file_path).getBlob().setName(name);
}

/* Куда складывается присланное: папка по месяцам на Диске, строка в журнале, письмо Олесе. */

function monthFolder(d) {
  var root = DriveApp.getFolderById(CFG.INBOX_FOLDER), name = Utilities.formatDate(d, CFG.TZ, 'yyyy-MM');
  var it = root.getFoldersByName(name);
  return it.hasNext() ? it.next() : root.createFolder(name);
}
function saveFile(fileId, name, d) { return monthFolder(d).createFile(downloadFile(fileId, name)); }

/* Сохранить одно сообщение родственника. Возвращает строку для письма или null. */
function saveMessage(m, uid, name, md) {
  var d = new Date(m.date * 1000), text = (m.text || '').trim();
  var base = Utilities.formatDate(d, CFG.TZ, 'yyyy-MM-dd_HHmmss') + '_' + (name || uid).split(' ')[0] + (md.person ? '_' + md.person : '') + '_' + m.message_id;  // номер сообщения, чтобы фото из альбома не путались
  var file = null, kind = 'текст', note = '';
  try {
    if (m.photo) { file = saveFile(m.photo[m.photo.length - 1].file_id, base + '.jpg', d); kind = 'фото'; }
    else if (m.document) {
      if ((m.document.file_size || 0) > CFG.MAX_FILE) note = 'Файл больше 20 МБ, бот не может его скачать';
      else { file = saveFile(m.document.file_id, base + '_' + (m.document.file_name || 'file'), d); kind = 'файл'; }
    }
    else if (m.voice || m.audio) { file = saveFile((m.voice || m.audio).file_id, base + '.ogg', d); kind = 'голосовое'; }
    else if (m.video || m.video_note) {
      var v = m.video || m.video_note;
      if ((v.file_size || 0) > CFG.MAX_FILE) { note = 'Видео больше 20 МБ'; say(uid, TXT.tooBig); }
      else { file = saveFile(v.file_id, base + '.mp4', d); kind = 'видео'; }
    }
    else if (!text) return null;   // стикеры и прочее не сохраняем
  } catch (e) { note = 'Не скачалось: ' + e; }

  var typeName = {photo: 'фото', story: 'рассказ', fix: 'неточность', answer: 'ответ на вопрос'}[md.kind] || '';
  var caption = (typeName ? '[' + typeName + (md.question ? ': ' + md.question : '') + '] ' : '') +
    (m.caption || text || '') + (note ? ' (' + note + ')' : '');
  journal([Utilities.formatDate(d, CFG.TZ, 'yyyy-MM-dd HH:mm'), name, file ? file.getUrl() : '', caption, '', md.about || '', '', '', 'новое', '']);
  return name + ': ' + kind + (typeName ? ' (' + typeName + ')' : '') + (caption ? ' ' + caption.slice(0, 200) : '');
}

// ===== Handlers.js =====
/* Главный цикл и ответы бота. */

/* Запускается триггером каждую минуту. */
function poll() {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(5000)) return;
  try {
    var st = loadState(), allow = loadAllow(), news = [];
    var upd = tg('getUpdates', {offset: st.offset, timeout: 0, allowed_updates: ['message', 'callback_query']});
    upd.forEach(function (u) {
      st.offset = u.update_id + 1;
      try { handle(u, st, allow, news); } catch (e) { console.log('update fail', e); }
    });
    setJ('state', st); setJ('allow', allow);
    if (news.length) notify(news);
  } finally { lock.releaseLock(); }
}

function handle(u, st, allow, news) {
  if (u.callback_query) return onDecision(u.callback_query, st, allow);
  var m = u.message;
  if (!m || m.chat.type !== 'private') return;
  var frm = m.from, uid = String(frm.id), text = (m.text || '').trim();

  if (claimAdmin(frm, uid, st, allow) && text.indexOf('/start') === 0) return;
  if (!allow[uid]) return askToJoin(frm, uid, st);
  if (onDeepLink(text, uid, st)) return;
  if (onCommand(text, uid, st)) return;

  var line = saveMessage(m, uid, allow[uid].name, st.mode[uid] || {});
  if (!line) return;
  news.push(line);
  var now = Date.now();
  if (now - (st.last[uid] || 0) > CFG.THANKS_EVERY) sayKb(uid, (st.mode[uid] || {}).kind === 'fix' ? TXT.thanksFix : TXT.thanks);
  st.last[uid] = now;
}

/* Олеся нажала «Пустить» или «Отклонить». */
function onDecision(q, st, allow) {
  if (String(q.from.id) !== String(st.admin)) return;
  var parts = q.data.split(':'), target = parts[1], p = st.pending[target];
  delete st.pending[target];
  var name = p ? p.name : target, txt;
  if (parts[0] === 'ok') { allow[target] = {name: name}; say(target, TXT.added); welcome(target); txt = 'Пустила: ' + name; }
  else txt = 'Отклонила: ' + name;
  try {
    tg('answerCallbackQuery', {callback_query_id: q.id, text: txt});
    tg('editMessageText', {chat_id: q.message.chat.id, message_id: q.message.message_id, text: txt});
  } catch (e) {}
}

/* Админа узнаём по нику. Возвращает true, если только что назначили. */
function claimAdmin(frm, uid, st, allow) {
  if ((frm.username || '').toLowerCase() !== CFG.ADMIN_USER || String(st.admin) === uid) return false;
  st.admin = uid; allow[uid] = {name: who(frm), admin: true};
  say(uid, TXT.adminHi); welcome(uid);
  return true;
}

function askToJoin(frm, uid, st) {
  if (st.pending[uid]) return;
  st.pending[uid] = {name: who(frm)};
  say(uid, TXT.wait);
  if (st.admin) say(st.admin, 'Хочет войти: ' + who(frm) + '\nid ' + uid,
    {reply_markup: {inline_keyboard: [[{text: 'Пустить', callback_data: 'ok:' + uid}, {text: 'Отклонить', callback_data: 'no:' + uid}]]}});
}

/* Кнопки «Рассказать» с сайта: ?start=a_<id>_<номер вопроса> и ?start=s_<id>. */
function onDeepLink(text, uid, st) {
  if (text.indexOf('/start ') !== 0) return false;
  var payload = text.slice(7).trim();
  if (payload.indexOf('a_') === 0) {
    var rest = payload.slice(2), cut = rest.lastIndexOf('_'), pid = rest.slice(0, cut), idx = rest.slice(cut + 1);
    var q = QUESTIONS.filter(function (x) { return x.id === pid && String(x.i) === idx; })[0];
    if (q) { askQuestion(uid, st, q, 'О ком: ' + q.who + '\nВопрос: ' + q.q + '\n\nРасскажите, что знаете. Можно текстом или голосовым, можно несколькими сообщениями.'); return true; }
  }
  if (payload.indexOf('s_') === 0 && PEOPLE[payload.slice(2)]) {
    var pid2 = payload.slice(2);
    st.mode[uid] = {kind: 'story', about: PEOPLE[pid2].full, person: pid2};
    sayKb(uid, 'История о человеке: ' + PEOPLE[pid2].full + '.\n\nЕсли вы написали текст на сайте, он уже скопирован: просто вставьте его сюда и отправьте. Или расскажите голосом.');
    return true;
  }
  return false;
}

function onCommand(text, uid, st) {
  if (text.indexOf('/start') === 0 || text === '/help') { welcome(uid); return true; }
  if (text === '/birthdays') { toggleMute(uid, st); return true; }
  if (text === '/fact' || text === BTN.FACT) { sayKb(uid, fact()); return true; }
  if (text === BTN.BOOK) { say(uid, TXT.book, {reply_markup: {inline_keyboard: [[{text: 'Открыть книгу', url: CFG.SITE}]]}}); return true; }
  if (text === BTN.PHOTO) { st.mode[uid] = {kind: 'photo'}; sayKb(uid, TXT.photo); return true; }
  if (text === BTN.STORY) { st.mode[uid] = {kind: 'story'}; sayKb(uid, TXT.story); return true; }
  if (text === BTN.FIX) { st.mode[uid] = {kind: 'fix'}; sayKb(uid, TXT.fix); return true; }
  if (text === BTN.Q) {
    var q = QUESTIONS[Math.floor(Math.random() * QUESTIONS.length)];
    askQuestion(uid, st, q, 'Вот что мы пока не знаем.\n\nО ком: ' + q.who + '\nВопрос: ' + q.q + '\n\nЕсли знаете, ответьте текстом или голосовым. Если нет, нажмите «Ответить на вопрос» ещё раз, будет другой вопрос.');
    return true;
  }
  return false;
}

function askQuestion(uid, st, q, text) {
  st.mode[uid] = {kind: 'answer', about: q.who, question: q.q, person: q.id};
  sayKb(uid, text);
}

function toggleMute(uid, st) {
  var i = st.mute.indexOf(uid);
  if (i >= 0) { st.mute.splice(i, 1); sayKb(uid, TXT.muteOff); }
  else { st.mute.push(uid); sayKb(uid, TXT.muteOn); }
}

function fact() { return 'А вы знали?\n\n' + FACTS[Math.floor(Math.random() * FACTS.length)]; }

function welcome(chat) {
  try { tg('sendPhoto', {chat_id: chat, photo: CFG.PHOTO, caption: TXT.intro, reply_markup: KB}); }
  catch (e) { sayKb(chat, TXT.intro); }
  sayKb(chat, fact());
}

// ===== Birthdays.js =====
/* Дни рождения живых и тихие дни памяти старших. Запускается триггером в 9:00 по Москве. */

function years(n) {
  var a = n % 10, b = n % 100;
  return n + ' ' + (a === 1 && b !== 11 ? 'год' : (a >= 2 && a <= 4 && (b < 10 || b >= 20)) ? 'года' : 'лет');
}

function todaysBirthdays(now) {
  now = now || new Date();
  var dm = Utilities.formatDate(now, CFG.TZ, 'dd.MM'), y = +Utilities.formatDate(now, CFG.TZ, 'yyyy');
  var day = +dm.slice(0, 2), mon = +dm.slice(3) - 1;
  return BIRTHS.filter(function (b) { return b.dm === dm; }).map(function (b) {
    if (b.memory) return '📖 В этот день, ' + day + ' ' + MONTHS[mon] + ' ' + b.y + ' года, ' +
      (b.g === 'f' ? 'родилась ' : 'родился ') + b.full + '.' + (b.note ? ' ' + b.note : '') +
      '\n\nИстория в книге: ' + CFG.SITE + '#p-' + b.id;
    return '🎂 Сегодня день рождения! ' + b.full + ', ' + b.short + '. Исполняется ' + years(y - b.y) + '.\n\nНе забудьте поздравить!';
  });
}

function birthdays() {
  var msgs = todaysBirthdays();
  if (!msgs.length) return;
  var st = loadState(), allow = loadAllow();
  Object.keys(allow).forEach(function (uid) {
    if (st.mute.indexOf(uid) >= 0) return;
    msgs.forEach(function (t) { say(uid, t); });
  });
}
